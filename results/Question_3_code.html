# Split the interests by comma, clean the labels, and count them
interest_counts <- interest %>%
  separate_rows(areas, sep = ",") %>%
  mutate(areas = str_remove(areas, ",?\\s*generally")) %>%
  mutate(areas = str_trim(areas)) %>%
  filter(!is.na(areas), areas != "", areas != "generally") %>%
  count(areas, sort = TRUE)

#What areas do students normally chose together? 
interest_long <- interest %>%
  mutate(student = row_number()) %>%
  separate_rows(areas, sep = ",") %>%
  mutate(areas = str_trim(areas)) %>%
  filter(!is.na(areas), areas != "", areas != "generally") %>%
  distinct(student, areas)

interest_binary <- interest_long %>%
  mutate(selected = 1) %>%
  pivot_wider(
    names_from = areas,
    values_from = selected,
    values_fill = 0
  )

interest_matrix <- interest_binary %>%
  select(-student)

set.seed(123)
clusters <- kmeans(interest_matrix, centers = 3)
interest_binary$cluster <- clusters$cluster

cluster_summary <- interest_binary %>%
  select(-student) %>%
  group_by(cluster) %>%
  summarise(across(everything(), mean), .groups = "drop")

pca <- prcomp(interest_matrix)
cluster_plot <- as.data.frame(pca$x[, 1:2])
cluster_plot$cluster <- as.factor(clusters$cluster)

ggplot(cluster_plot, aes(x = PC1, y = PC2, color = cluster)) +
  geom_point(size = 3) +
  labs(
    title = "Student Clusters Based on Data Science Interests",
    x = "Principal Component 1",
    y = "Principal Component 2",
    color = "Cluster"
  ) +
  theme_minimal()
#Cluster 1 Red: Students who commonly selected deep learning and image analysis.
#Cluster 2 Green: Students who commonly selected databases, deep learning, and model deployment
#Cluster 3 Blue: Students who commonly selected statistics, visualization, and data acquisition
#What do the clusters mean 
cluster_graph <- cluster_summary %>%
  pivot_longer(
    cols = -cluster,
    names_to = "Interest",
    values_to = "Percentage"
  )
ggplot(cluster_graph, aes(x = factor(cluster), y = Interest,
                          fill = Percentage)) +
  geom_tile(color = "black") +
  geom_text(aes(label = paste0(round(Percentage * 100), "%")),
            size = 3) +
  scale_fill_gradient(low = "white", high = "steelblue") +
  labs(
    title = "Data Science Interests by Cluster",
    x = "Cluster",
    y = "Data Science Interest",
    fill = "Proportion"
  ) +
  theme_minimal()
